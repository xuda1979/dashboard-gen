function validateFetchUrl(url) {
  if (!/^[a-zA-Z][a-zA-Z0-9+.-]*:/.test(url)) {
    return; // e.g. market.json
  }
  let parsed;
  try {
    parsed = new URL(url);

    throw new FetchSpecError(`Invalid fetch URL "${url}". Use a relative path, absolute http(s) URL, or "self://examples/<file>.json".`, 'data.fetch');
  }
  const scheme = parsed.protocol.replace(/:$/, '');
  const isHttp = scheme === 'http' || scheme === 'https';
  if (!isHttp && scheme !== 'self') {
    throw new FetchSpecError(
      `Fetch URL scheme "${scheme}" not supported. Use http(s) or self://examples/<file>.json`,
      'data.fetch'
    );
  }
  const host = (parsed.hostname || '').toLowerCase();
  if (isHttp && (!host || !host.includes('.'))) {
    throw new FetchSpecError(`Fetch URL must include a valid hostname (got "${url}")`, 'data.fetch');
  }
}

function loadDataFile(relPath, fileDir, at) {
  const { DataFileError } = require('./errors');
  const fs = require('node:fs');
  const path = require('node:path');
  if (!relPath) throw new DataFileError('chart "file" must be a non-empty path', at);
  const abs = path.isAbsolute(relPath) ? relPath : path.join(fileDir || '.', relPath);
  let content;
  try {
    content = fs.readFileSync(abs, 'utf8');
  } catch (err) {
    throw new DataFileError(`Could not read data file "${relPath}" (${err.code || err.message})`, at);
  }
  const ext = path.extname(abs).toLowerCase();
  if (ext === '.json') {
    try {
      return JSON.parse(content);
    } catch (err) {
      throw new DataFileError(`Data file "${relPath}" is not valid JSON: ${err.message}`, at);
    }
  }
  if (ext === '.csv') return csvToRows(content);
  throw new DataFileError(`Unsupported data file extension "${ext}" (use .json or .csv).`, at);
}
