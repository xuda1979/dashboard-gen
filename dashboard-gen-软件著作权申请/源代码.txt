class DashboardError extends Error {
  constructor(message, path) {
    super(message);
    this.name = this.constructor.name;
    this.path = path || '';
  }
  format() {
    const where = this.path ? `\n  at: ${this.path}` : '';
    return `${this.message}${where}`;
  }
}
class StyleError extends DashboardError {}
class FetchSpecError extends DashboardError {}
class DataFileError extends DashboardError {}
class ParseError extends DashboardError {}
class NotImplementedError extends DashboardError {}

function validateFetchUrl(url) {
  if (!/^[a-zA-Z][a-zA-Z0-9+.-]*:/.test(url)) {
    return; // e.g. market.json
  }
  let parsed;

    throw new FetchSpecError(`Invalid fetch URL "${url}". Use a relative path, absolute http(s) URL, or "self://examples/<file>.json".`, 'data.fetch');
  }
  const scheme = parsed.protocol.replace(/:$/, '');
  const isHttp = scheme === 'http' || scheme === 'https';
  if (!isHttp && scheme !== 'self') {
    throw new FetchSpecError(
      `Fetch URL scheme "${scheme}" not supported. Use http(s) or self://examples/<file>.json`,
      'data.fetch'
    );
  }
  const host = (parsed.hostname || '').toLowerCase();
  if (isHttp && (!host || !host.includes('.'))) {
    throw new FetchSpecError(`Fetch URL must include a valid hostname (got "${url}")`, 'data.fetch');
  }

  }

function parseArgs(argv) {
  const out = { _: [] };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a.startsWith('--')) {
      const eq = a.indexOf('=');
      if (eq > 0) {
        out[a.slice(2, eq)] = a.slice(eq + 1);
      } else {
        const key = a.slice(2);
        const next = argv[i + 1];
        if (next !== undefined && !next.startsWith('-')) {
          out[key] = next;
          i++;
        } else {
          out[key] = true;
        }
      }
    } else if (a.startsWith('-') && a.length > 1) {
      const key = a.slice(1);
      const next = argv[i + 1];
      if (next !== undefined && !next.startsWith('-')) {
        out[key] = next;
        i++;
      } else {
        out[key] = true;
      }
    } else {
      out._.push(a);
    }
  }
  return out;
}
